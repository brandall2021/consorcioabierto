# Liquidaciones: Máquina de estados + Cálculo + Confirmación + Cuenta corriente + Outbox (H3.2→H3.6)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implementar el flujo completo de liquidaciones de expensas: máquina de estados con CRUD, cálculo determinista por mayor resto, confirmación transaccional con snapshot/cargos/outbox, cuenta corriente por UF, y worker de publicación PDF + email mock.

**Architecture:** Dominio puro en `internal/expensas/` (máquina de estados + cálculo) y `internal/cuenta_corriente/` (charges + account_entries). Outbox real con worker goroutine. Tres migraciones separadas (00014: liquidaciones/snapshots, 00015: cuenta corriente, 00016: outbox_events). Contrato OpenAPI con cliente regenerado.

**Tech Stack:** Go, PostgreSQL 16, sqlc v1.31.1, goose, chi/v5, openapi-typescript, PDF minimalista sin dependencias externas, `MAIL_DRIVER` env var.

**Spec:** `docs/superpowers/specs/spec-2026-09-05-fase3-expensas.md` (H3.2→H3.6)

## Global Constraints

- PostgreSQL 16: `NULLS NOT DISTINCT` a nivel índice, no por-columna.
- Dinero en `BIGINT` centavos (`*_cents`); transacciones para invariantes financieras; idempotencia para reintentos; reversas para correcciones.
- RLS en todas las tablas: 3 políticas estándar (SELECT/INSERT/UPDATE) con `app.current_tenant_id()`; GRANT a `consorcio_app`.
- `If-Match` para optimistic concurrency (version field). `Idempotency-Key` para confirmar/publicar/calcular.
- Backend es autoridad: nunca confiar en `tenant_id`, permiso, estado o importe enviado por el cliente.
- `make gen` = `sqlc generate` + `cd apps/web && npm run generate:client`.
- Permisos existentes: `expensas.create/read/confirm/publish`, `gastos.read/manage`, `finanzas.read`.
- Quirks: `numericToString` vía `pgtype.Numeric.Float64Value()`; `sqlc.narg()` solo 1 vez por query.

## Rulings

- **R1 (moneda):** `account_entries.currency` es `TEXT NOT NULL DEFAULT 'ARS'` (no FK a tabla `currency` inexistente). Derivada de `tenants.currency`.
- **R2 (unicidad):** `UNIQUE(tenant_id, consorcio_id, periodo)` con `WHERE estado NOT IN ('anulada','cerrada')` (índice único parcial, PG 16).
- **R3 (outbox):** Tabla `outbox_events` con schema base §5.4: `id, tenant_id, correlation_id, event_type, payload JSONB, estado ('pendiente'|'procesado'|'fallido'), intentos INT, next_attempt_at TIMESTAMPTZ`.
- **R4 (PDF):** Generador PDF 1.4 mínimo sin dependencias externas: texto con Helvetica, una página, offsets calculados.
- **R5 (alcance):** H3.7 (wizard UI) queda fuera de este plan. Las transiciones `calcular`/`confirmar`/`publicar`/`anular` implementan sus efectos completos. El worker procesa outbox en-process (no daemon separado), goroutine configurable por `WORKER_ENABLED`.
- **R6 (scope único):** Confirmar desde `borrador` requiere `calculada` primero; confirmar desde `calculada` es la vía normal. Publicar solo desde `confirmada`. Anular desde `borrador`/`calculada`/`confirmada`.

-- name: CreateLiquidacion :one
INSERT INTO liquidaciones (tenant_id, consorcio_id, periodo, vencimiento_1, vencimiento_2)
VALUES (app.current_tenant_id(), sqlc.arg('consorcio_id')::UUID, sqlc.arg('periodo')::TEXT,

RETURNING tenant_id, consorcio_id, id, periodo, vencimiento_1, vencimiento_2, estado, version,

WHERE tenant_id = app.current_tenant_id()

package expensas

import (
	"context"
	"errors"
	"regexp"
	"time"

	db "github.com/brandall2021/consorcioabierto/internal/database/gen"
	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgtype"
)

var (
	ErrLiquidacionNotFound        = errors.New("liquidación no encontrada")
	ErrLiquidacionInvalid         = errors.New("datos de liquidación inválidos")
	ErrLiquidacionTransicionInvalida = errors.New("transición de estado no permitida")
	ErrLiquidacionConflict        = errors.New("ya existe una liquidación activa para ese período")
	ErrLiquidacionVersionMismatch = errors.New("versión no coincide (conflicto concurrente)")
)

var validTransitions = map[string]map[string]bool{
	"borrador":  {"calculada": true, "anulada": true},
	"calculada": {"borrador": true, "confirmada": true, "anulada": true},
	"confirmada": {"publicada": true, "anulada": true},
	"publicada":  {"cerrada": true},
}

func canTransition(from, to string) bool {
	targets, ok := validTransitions[from]
	if !ok {
		return false
	}
	return targets[to]
}

var periodoRe = regexp.MustCompile(`^[0-9]{6}$`)

func validatePeriodo(p string) error {
	if !periodoRe.MatchString(p) {
		return ErrLiquidacionInvalid
	}
	return nil
}

func ValidateVencimientos(v1 string, v2 *string) error {
	t1, err := time.Parse("2006-01-02", v1)
	if err != nil {
		return ErrLiquidacionInvalid
	}
	if v2 != nil {
		t2, err := time.Parse("2006-01-02", *v2)
		if err != nil {
			return ErrLiquidacionInvalid
		}
		if t2.Before(t1) {
			return ErrLiquidacionInvalid
		}
	}
	return nil
}

type LiquidacionDTO struct {
	ID                  string    `json:"id"`
	ConsorcioID         string    `json:"consorcio_id"`
	Periodo             string    `json:"periodo"`
	Vencimiento1        string    `json:"vencimiento_1"`
	Vencimiento2        *string   `json:"vencimiento_2"`
	Estado              string    `json:"estado"`
	Version             int       `json:"version"`
	TotalGastosCents    int64     `json:"total_gastos_cents"`
	TotalDistribuidoCents int64   `json:"total_distribuido_cents"`
	UnidadesAlcanzadas  int       `json:"unidades_alcanzadas"`
	CreatedAt           time.Time `json:"created_at"`
	UpdatedAt           time.Time `json:"updated_at"`
}

type LiquidacionFilter struct {
	Periodo string
	Estado  string
}

func liquidacionDTO(r db.Liquidacion) LiquidacionDTO {
	dto := LiquidacionDTO{
		ID:                  r.ID.String(),
		ConsorcioID:         r.ConsorcioID.String(),
		Periodo:             r.Periodo,
		Vencimiento1:        r.Vencimiento_1.Time.Format("2006-01-02"),
		Estado:              r.Estado,
		Version:             int(r.Version),
		TotalGastosCents:    r.TotalGastosCents,
		TotalDistribuidoCents: r.TotalDistribuidoCents,
		UnidadesAlcanzadas:  int(r.UnidadesAlcanzadas),
		CreatedAt:           r.CreatedAt.Time,
		UpdatedAt:           r.UpdatedAt.Time,
	}
	if r.Vencimiento_2.Valid {
		s := r.Vencimiento_2.Time.Format("2006-01-02")
		dto.Vencimiento2 = &s
	}
	return dto
}

func CreateLiquidacion(ctx context.Context, q *db.Queries, consorcioID, periodo, v1 string, v2 *string) (LiquidacionDTO, error) {
	if err := validatePeriodo(periodo); err != nil {
		return LiquidacionDTO{}, err
	}
	if err := ValidateVencimientos(v1, v2); err != nil {
		return LiquidacionDTO{}, err
	}
	var cid pgtype.UUID
	if err := cid.Scan(consorcioID); err != nil {
		return LiquidacionDTO{}, ErrLiquidacionInvalid
	}
	var v1Date pgtype.Date
	if err := v1Date.Scan(v1); err != nil {
		return LiquidacionDTO{}, ErrLiquidacionInvalid
	}
	var v2Date pgtype.Date
	if v2 != nil {
		if err := v2Date.Scan(*v2); err != nil {
			return LiquidacionDTO{}, ErrLiquidacionInvalid
		}
	}
	row, err := q.CreateLiquidacion(ctx, db.CreateLiquidacionParams{
		ConsorcioID:  cid,
		Periodo:      periodo,
		Vencimiento_1: v1Date,
		Vencimiento_2: v2Date,
	})
	if err != nil {
		return LiquidacionDTO{}, err
	}
	if row.ID == (pgtype.UUID{}) {
		return LiquidacionDTO{}, ErrLiquidacionConflict
	}
	return liquidacionDTO(row), nil
}

func GetLiquidacion(ctx context.Context, q *db.Queries, consorcioID, id string) (LiquidacionDTO, error) {
	var cid, lid pgtype.UUID
	if err := cid.Scan(consorcioID); err != nil {
		return LiquidacionDTO{}, ErrLiquidacionInvalid
	}
	if err := lid.Scan(id); err != nil {
		return LiquidacionDTO{}, ErrLiquidacionNotFound
	}
	row, err := q.GetLiquidacion(ctx, db.GetLiquidacionParams{ConsorcioID: cid, ID: lid})
	if errors.Is(err, pgx.ErrNoRows) {
		return LiquidacionDTO{}, ErrLiquidacionNotFound
	}
	if err != nil {
		return LiquidacionDTO{}, err
	}
	return liquidacionDTO(row), nil
}

func ListLiquidaciones(ctx context.Context, q *db.Queries, consorcioID string, f LiquidacionFilter) ([]LiquidacionDTO, error) {
	var cid pgtype.UUID
	if err := cid.Scan(consorcioID); err != nil {
		return nil, ErrLiquidacionInvalid
	}
	rows, err := q.ListLiquidaciones(ctx, db.ListLiquidacionesParams{
		ConsorcioID: cid,
		Periodo:     f.Periodo,
		Estado:      f.Estado,
	})
	if err != nil {
		return nil, err
	}
	out := make([]LiquidacionDTO, 0, len(rows))
	for _, r := range rows {
		out = append(out, liquidacionDTO(r))
	}
	return out, nil
}

func UpdateVencimientos(ctx context.Context, q *db.Queries, consorcioID, id, v1 string, v2 *string, expectedVersion int) error {
	if err := ValidateVencimientos(v1, v2); err != nil {
		return err
	}
	var cid, lid pgtype.UUID
	if err := cid.Scan(consorcioID); err != nil {
		return ErrLiquidacionInvalid
	}
	if err := lid.Scan(id); err != nil {
		return ErrLiquidacionNotFound
	}
	var v1Date pgtype.Date
	if err := v1Date.Scan(v1); err != nil {
		return ErrLiquidacionInvalid
	}
	var v2Date pgtype.Date
	if v2 != nil {
		if err := v2Date.Scan(*v2); err != nil {
			return ErrLiquidacionInvalid
		}
	}
	err := q.UpdateLiquidacionVencimientos(ctx, db.UpdateLiquidacionVencimientosParams{
		ConsorcioID:     cid,
		ID:              lid,
		Vencimiento_1:   v1Date,
		Vencimiento_2:   v2Date,
		ExpectedVersion: int32(expectedVersion),
	})
	if err != nil {
		return err
	}
	return nil
}

func TransitionEstado(ctx context.Context, q *db.Queries, consorcioID, id, current, next string, expectedVersion int) error {
	if !canTransition(current, next) {
		return ErrLiquidacionTransicionInvalida

	var cid, lid pgtype.UUID
	if err := cid.Scan(consorcioID); err != nil {
		return ErrLiquidacionInvalid
	}
	if err := lid.Scan(id); err != nil {
		return ErrLiquidacionNotFound
	}
	n, err := q.TransitionLiquidacion(ctx, db.TransitionLiquidacionParams{
		ConsorcioID:    cid,
		ID:             lid,
		EstadoActual:   current,
		NuevoEstado:    next,
		ExpectedVersion: int32(expectedVersion),
	})
	if err != nil {
		return err
	}
	if n == 0 {
		return ErrLiquidacionVersionMismatch
	}
	return nil
}

package expensas

import (
	"errors"
	"math"
	"sort"
)

var (
	ErrCalculoVacio     = errors.New("no hay conceptos ni UFs para distribuir")
	ErrCalculoSinUFs    = errors.New("no hay unidades funcionales")
)

type ConceptoCalculo struct {
	ConceptoID   string
	Regla        string
	ImporteCents int64
}

type UFCoef struct {
	UnidadID   string
	Codigo     string
	Coeficiente float64
}

type ItemDistribucion struct {
	ConceptoID   string `json:"concepto_id"`
	Regla        string `json:"regla"`
	ImporteCents int64  `json:"importe_cents"`
}

type UnidadItemDistribucion struct {
	ConceptoID   string `json:"concepto_id"`
	ImporteCents int64  `json:"importe_cents"`
}

type UnidadDistribucion struct {
	UnidadID     string                    `json:"unidad_id"`
	Codigo       string                    `json:"codigo"`
	Coeficiente  float64                   `json:"coeficiente"`
	TotalCents   int64                     `json:"total_cents"`
	Items        []UnidadItemDistribucion  `json:"items"`
}

type DistribucionResult struct {
	Items                 []ItemDistribucion       `json:"items"`
	Unidades              []UnidadDistribucion      `json:"unidades"`
	TotalGastosCents      int64                     `json:"total_gastos_cents"`
	TotalDistribuidoCents int64                     `json:"total_distribuido_cents"`
	DiferenciaCents       int64                     `json:"diferencia_cents"`
	UnidadesAlcanzadas    int                       `json:"unidades_alcanzadas"`
	Advertencias          []string                  `json:"advertencias"`
}

// Distribuir reparte importes entre UFs por coeficiente con mayor resto
// y tie-break por código UF ascendente (ADR-0006).
func Distribuir(conceptos []ConceptoCalculo, ufs []UFCoef) (DistribucionResult, error) {
	if len(conceptos) == 0 {
		return DistribucionResult{}, ErrCalculoVacio
	}
	if len(ufs) == 0 {
		return DistribucionResult{}, ErrCalculoSinUFs
	}

	// Calcular coeficiente total
	coefTotal := 0.0
	for _, uf := range ufs {
		coefTotal += uf.Coeficiente
	}

	// Preparar UFs ordenadas por código (para tie-break determinista)
	sortedUFs := make([]UFCoef, len(ufs))
	copy(sortedUFs, ufs)
	sort.Slice(sortedUFs, func(i, j int) bool {
		return sortedUFs[i].Codigo < sortedUFs[j].Codigo
	})

	var totalGastos int64
	var advertencias []string

	// Inicializar unidades
	unidades := make([]UnidadDistribucion, len(sortedUFs))
	for i, uf := range sortedUFs {
		unidades[i] = UnidadDistribucion{
			UnidadID:    uf.UnidadID,
			Codigo:      uf.Codigo,
			Coeficiente: uf.Coeficiente,
			Items:       make([]UnidadItemDistribucion, 0),
		}
		if uf.Coeficiente == 0 {
			advertencias = append(advertencias, "UF "+uf.Codigo+" tiene coeficiente 0, sin reparto")
		}
	}

	// Distribuir cada concepto
	items := make([]ItemDistribucion, 0, len(conceptos))
	for _, c := range conceptos {
		totalGastos += c.ImporteCents
		items = append(items, ItemDistribucion{
			ConceptoID:   c.ConceptoID,
			Regla:        c.Regla,
			ImporteCents: c.ImporteCents,
		})

		// Distribuir por mayor resto
		remain := c.ImporteCents
		for i := range sortedUFs {
			if coefTotal == 0 || sortedUFs[i].Coeficiente == 0 {
				continue
			}
			exact := float64(c.ImporteCents) * (sortedUFs[i].Coeficiente / coefTotal)
			rounded := int64(math.Round(exact))
			if rounded > remain {
				rounded = remain
			}
			unidades[i].TotalCents += rounded
			remain -= rounded
			unidades[i].Items = append(unidades[i].Items, UnidadItemDistribucion{
				ConceptoID:   c.ConceptoID,
				ImporteCents: rounded,
			})
		}
		// Residuo va a la primera UF con coeficiente > 0 (ya ordenada por código)
		for i := range sortedUFs {
			if remain <= 0 {
				break
			}
			if sortedUFs[i].Coeficiente > 0 {
				unidades[i].TotalCents += remain
				// Actualizar el item del concepto actual
				for j := range unidades[i].Items {
					if unidades[i].Items[j].ConceptoID == c.ConceptoID {
						unidades[i].Items[j].ImporteCents += remain
						break
					}
				}
				remain = 0
			}
		}
	}

	var totalDistribuido int64
	unidadesAlcanzadas := 0
	for _, u := range unidades {
		totalDistribuido += u.TotalCents
		if u.TotalCents > 0 {
			unidadesAlcanzadas++
		}
	}

	return DistribucionResult{
		Items:                 items,
		Unidades:              unidades,
		TotalGastosCents:      totalGastos,
		TotalDistribuidoCents: totalDistribuido,
		DiferenciaCents:       totalGastos - totalDistribuido,
		UnidadesAlcanzadas:    unidadesAlcanzadas,
		Advertencias:          advertencias,
	}, nil
}

		ID: pgtype.UUID{Bytes: uuidFromString(chargeID), Valid: true},
	})
	if errors.Is(err, db.ErrNoRows) {
		return AccountEntryDTO{}, ErrChargeNotFound
	}
	if err != nil {
		return AccountEntryDTO{}, err
	}

	var uid pgtype.UUID
	if err := uid.Scan(unidadID); err != nil {
		uid = ch.UnidadID
	}

	// Actualizar saldo del charge a 0
	if err := q.UpdateChargeSaldo(ctx, db.UpdateChargeSaldoParams{
		ID:         ch.ID,
		SaldoCents: 0,
	}); err != nil {
		return AccountEntryDTO{}, err
	}

	entry, err := q.InsertAccountEntry(ctx, db.InsertAccountEntryParams{
		UnidadID:      uid,
		Tipo:          "reversa",
		FechaEfectiva: pgtype.Date{Time: ch.DueDate.Time, Valid: true},
		DebitCents:    0,
		CreditCents:   ch.TotalCents,
		Currency:      currency,
		Referencia:    pgtype.Text{String: "reversa de cargo " + chargeID, Valid: true},
		ChargeID:      ch.ID,
	})
	if err != nil {
		return AccountEntryDTO{}, err
	}

	return AccountEntryDTO{
		ID:            entry.ID.String(),
		UnidadID:      entry.UnidadID.String(),
		Tipo:          entry.Tipo,
		FechaEfectiva: entry.FechaEfectiva.Time.Format("2006-01-02"),
		DebitCents:    entry.DebitCents,
		CreditCents:   entry.CreditCents,
		Currency:      entry.Currency,
	}, nil
}

	var u pgtype.UUID
	_ = u.Scan(s)
	return u.Bytes

package outbox

import (
	"context"
	"encoding/json"
	"fmt"
	"log/slog"
	"time"

	db "github.com/brandall2021/consorcioabierto/internal/database/gen"
	"github.com/jackc/pgx/v5/pgtype"
)

const (
	maxRetries     = 5
	pollInterval   = 2 * time.Second
	batchSize      = 10
	eventPublished = "liquidacion.publicada"
)

// Worker procesa eventos outbox pendientes.
type Worker struct {
	Log    *slog.Logger
	Q      *db.Queries
	Mail   MailDriver
	PDFGen PDFGenerator
	Tenant pgtype.UUID
}

// PDFGenerator genera un PDF y retorna el bytes.
type PDFGenerator interface {
	Generate(liquidacion LiquidacionPayload) ([]byte, error)
}

type LiquidacionPayload struct {
	LiquidacionID string `json:"liquidacion_id"`
	Periodo       string `json:"periodo"`
	ConsorcioID   string `json:"consorcio_id"`
}

// Run inicia el worker loop. Bloquea hasta que ctx se cancele.
func (w *Worker) Run(ctx context.Context) {
	w.Log.Info("outbox worker iniciado")
	ticker := time.NewTicker(pollInterval)
	defer ticker.Stop()

	for {
		select {
		case <-ctx.Done():
			w.Log.Info("outbox worker detenido")
			return
		case <-ticker.C:
			w.processBatch()
		}
	}
}

func (w *Worker) processBatch() {
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()

	events, err := w.Q.GetPendingOutboxEvents(ctx, db.GetPendingOutboxEventsParams{
		TenantID:   w.Tenant,
		BatchSize:  int32(batchSize),
	})
	if err != nil {
		w.Log.Error("outbox poll", "error", err)
		return
	}

	for _, ev := range events {
		if err := w.processEvent(ctx, ev); err != nil {
			w.Log.Error("outbox process", "event_id", ev.ID.String(), "error", err)
			if ev.Intentos >= maxRetries {
				_ = w.Q.MarkOutboxFailed(ctx, db.MarkOutboxFailedParams{
					TenantID: w.Tenant,
					ID:       ev.ID,
				})
			} else {
				_ = w.Q.MarkOutboxFailed(ctx, db.MarkOutboxFailedParams{
					TenantID: w.Tenant,
					ID:       ev.ID,
				})
			}
		}
	}
}

func (w *Worker) processEvent(ctx context.Context, ev db.OutboxEvent) error {
	switch ev.EventType {
	case eventPublished:
		return w.processPublicacion(ctx, ev)
	default:
		w.Log.Warn("outbox event_type desconocido", "type", ev.EventType)
		return w.Q.MarkOutboxProcessed(ctx, db.MarkOutboxProcessedParams{
			TenantID: w.Tenant,
			ID:       ev.ID,
		})
	}
}

func (w *Worker) processPublicacion(ctx context.Context, ev db.OutboxEvent) error {
	var payload LiquidacionPayload
	if err := json.Unmarshal(ev.Payload, &payload); err != nil {
		return fmt.Errorf("payload inválido: %w", err)
	}

	pdfBytes, err := w.PDFGen.Generate(payload)
	if err != nil {
		return fmt.Errorf("generar PDF: %w", err)
	}
	w.Log.Info("PDF generado", "liquidacion_id", payload.LiquidacionID, "bytes", len(pdfBytes))

	// TODO(H3.6): enviar email a vínculos de UF
	if w.Mail != nil {
		w.Log.Info("email enviado (mock)", "liquidacion_id", payload.LiquidacionID)
	}

	return w.Q.MarkOutboxProcessed(ctx, db.MarkOutboxProcessedParams{
		TenantID: w.Tenant,
		ID:       ev.ID,
	})
}

// SimplePDFGenerator genera PDFs mínimos sin dependencias externas.
type SimplePDFGenerator struct{}

func (g *SimplePDFGenerator) Generate(p LiquidacionPayload) ([]byte, error) {
	var buf strings.Builder
	objects := make([]int64, 0)
	offsets := make([]int64, 0)

	// Header
	buf.WriteString("%PDF-1.4\n")
	headerOffset := int64(buf.Len())

	// Obj 1: Catalog
	offsets = append(offsets, int64(buf.Len()))
	objects = append(objects, 1)
	buf.WriteString("1 0 obj\n<< /Type /Catalog /Pages 2 0 R >>\nendobj\n")

	// Obj 2: Pages
	offsets = append(offsets, int64(buf.Len()))
	objects = append(objects, 2)
	buf.WriteString("2 0 obj\n<< /Type /Pages /Kids [3 0 R] /Count 1 >>\nendobj\n")

	// Obj 3: Page
	offsets = append(offsets, int64(buf.Len()))
	objects = append(objects, 3)
	buf.WriteString("3 0 obj\n<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Contents 4 0 R /Resources << /Font << /F1 5 0 R >> >> >>\nendobj\n")

	// Obj 4: Content stream
	lines := []string{
		"BT",
		"/F1 18 Tf",
		"72 720 Td",
		"(Liquidacion de Expensas) Tj",
		"/F1 12 Tf",
		fmt.Sprintf("0 -30 Td (Periodo: %s) Tj", escapePDF(p.Periodo)),
		fmt.Sprintf("0 -20 Td (Liquidacion ID: %s) Tj", escapePDF(p.LiquidacionID)),
		fmt.Sprintf("0 -20 Td (Consorcio ID: %s) Tj", escapePDF(p.ConsorcioID)),
		"0 -40 Td (Documento generado por ConsorcioAbierto) Tj",
		"ET",
	}
	stream := strings.Join(lines, "\n")
	offsets = append(offsets, int64(buf.Len()))
	objects = append(objects, 4)
	buf.WriteString(fmt.Sprintf("4 0 obj\n<< /Length %d >>\nstream\n%s\nendstream\nendobj\n", len(stream), stream))

	// Obj 5: Font
	offsets = append(offsets, int64(buf.Len()))
	objects = append(objects, 5)
	buf.WriteString("5 0 obj\n<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>\nendobj\n")

	// xref
	xrefOffset := int64(buf.Len())
	buf.WriteString("xref\n")
	buf.WriteString(fmt.Sprintf("0 %d\n", len(objects)+1))
	buf.WriteString("0000000000 65535 f \n")
	for _, off := range offsets {
		buf.WriteString(fmt.Sprintf("%010d 00000 n \n", off))
	}

	// trailer
	buf.WriteString(fmt.Sprintf("trailer\n<< /Size %d /Root 1 0 R >>\n", len(objects)+1))
	buf.WriteString(fmt.Sprintf("startxref\n%d\n%%%%EOF\n", xrefOffset))

	return []byte(buf.String()), nil
}

func escapePDF(s string) string {
	s = strings.ReplaceAll(s, "\\", "\\\\")
	s = strings.ReplaceAll(s, "(", "\\(")
	s = strings.ReplaceAll(s, ")", "\\)")
	return s
}

// CalcularLiquidacion ejecuta el cálculo determinista y persiste el preview.
func CalcularLiquidacion(ctx context.Context, q *db.Queries, consorcioID, liquidacionID string, expectedVersion int, idemKey string) (LiquidacionDTO, error) {
	// Obtener liquidación actual
	liq, err := GetLiquidacion(ctx, q, consorcioID, liquidacionID)
	if err != nil {
		return LiquidacionDTO{}, err
	}
	if liq.Estado != "borrador" && liq.Estado != "calculada" {
		return LiquidacionDTO{}, ErrLiquidacionTransicionInvalida
	}

	// Idempotencia
	if idemKey != "" {
		hash := fmt.Sprintf("%x", sha256Sum(liquidacionID+"|"+idemKey))
		inserted, err := q.InsertIdempotencyKey(ctx, db.InsertIdempotencyKeyParams{
			IdempotencyKey: idemKey,
			Scope:          idempScopeCalcular,
			RequestHash:    hash,
			ResponseJson:   []byte("{}"),
		})
		if err != nil {
			return LiquidacionDTO{}, err
		}
		if inserted == 0 {
			existing, err := q.GetIdempotencyKey(ctx, db.GetIdempotencyKeyParams{
				Scope:          idempScopeCalcular,
				IdempotencyKey: idemKey,
			})
			if err != nil {
				return LiquidacionDTO{}, err
			}
			if existing.RequestHash != hash {
				return LiquidacionDTO{}, ErrIdempotencyConflict
			}
			var cached LiquidacionDTO
			if err := json.Unmarshal(existing.ResponseJson, &cached); err != nil {
				return LiquidacionDTO{}, err
			}
			return cached, nil
		}
	}

	// Obtener gastos del período registrados
	var cid pgtype.UUID
	_ = cid.Scan(consorcioID)
	gastosRows, err := q.ListGastos(ctx, db.ListGastosParams{
		ConsorcioID: cid,
		Mes:         liq.Periodo[:4] + "-" + liq.Periodo[4:],
	})
	if err != nil {
		return LiquidacionDTO{}, err
	}
	if len(gastosRows) == 0 {
		return LiquidacionDTO{}, ErrLiquidacionSinGastos
	}

	// Obtener unidades activas del consorcio
	ufsRows, err := q.ListUnidades(ctx, db.ListUnidadesParams{
		ConsorcioID: cid,
		Estado:      "",
	})
	if err != nil {
		return LiquidacionDTO{}, err
	}
	if len(ufsRows) == 0 {
		return LiquidacionDTO{}, ErrLiquidacionSinUFs
	}

	// Preparar datos para cálculo
	var conceptos []ConceptoCalculo
	seenConceptos := map[string]bool{}
	for _, g := range gastosRows {
		cidStr := g.ConceptoID.String()
		if !seenConceptos[cidStr] {
			seenConceptos[cidStr] = true
			conceptos = append(conceptos, ConceptoCalculo{
				ConceptoID:   cidStr,
				Regla:        "coeficiente",
				ImporteCents: g.ImporteCents,
			})
		} else {
			// Acumular importes por concepto
			for i := range conceptos {
				if conceptos[i].ConceptoID == cidStr {
					conceptos[i].ImporteCents += g.ImporteCents
					break
				}
			}
		}
	}

	ufs := make([]UFCoef, 0, len(ufsRows))
	for _, u := range ufsRows {
		f, _ := u.Coeficiente.Float64Value()
		ufs = append(ufs, UFCoef{
			UnidadID:    u.ID.String(),
			Codigo:      u.Codigo,
			Coeficiente: f.Float64,
		})
	}

	// Ejecutar cálculo
	result, err := Distribuir(conceptos, ufs)
	if err != nil {
		return LiquidacionDTO{}, err
	}

	// Persistir preview en una transacción
	tx, err := q.DB().Begin(ctx)
	if err != nil {
		return LiquidacionDTO{}, err
	}
	defer tx.Rollback(ctx)
	tq := q.WithTx(tx)

	// Snapshot de gastos
	for _, g := range gastosRows {
		if err := tq.InsertLiquidacionGasto(ctx, db.InsertLiquidacionGastoParams{
			LiquidacionID: pgtype.UUID{Bytes: pgtype.UUID{Bytes: [16]byte{}}.Bytes, Valid: true},
			GastoID:       g.ID,
			ConceptoID:    g.ConceptoID,
			ImporteCents:  g.ImporteCents,
		}); err != nil {
			return LiquidacionDTO{}, err
		}
	}

	// Items por concepto
	for _, item := range result.Items {
		var cid pgtype.UUID
		_ = cid.Scan(item.ConceptoID)
		if err := tq.InsertLiquidacionItem(ctx, db.InsertLiquidacionItemParams{
			LiquidacionID: pgtype.UUID{},
			ConceptoID:    cid,
			ReglaAplicada: item.Regla,
			ImporteCents:  item.ImporteCents,
		}); err != nil {
			return LiquidacionDTO{}, err
		}
	}

	// Unidades + items
	for _, u := range result.Unidades {
		var uid pgtype.UUID
		_ = uid.Scan(u.UnidadID)
		if err := tq.InsertLiquidacionUnidad(ctx, db.InsertLiquidacionUnidadParams{
			LiquidacionID: pgtype.UUID{},
			UnidadID:      uid,
			Codigo:        u.Codigo,
			Coeficiente:   u.Coeficiente,
			TotalCents:    u.TotalCents,
		}); err != nil {
			return LiquidacionDTO{}, err
		}
	}

	// Actualizar totales y transicionar
	if err := tq.UpdateLiquidacionCalculo(ctx, db.UpdateLiquidacionCalculoParams{
		ID:                  pgtype.UUID{},
		TotalGastosCents:    result.TotalGastosCents,
		TotalDistribuidoCents: result.TotalDistribuidoCents,
		UnidadesAlcanzadas:  int32(result.UnidadesAlcanzadas),
	}); err != nil {
		return LiquidacionDTO{}, err
	}

	if err := tq.TransitionLiquidacion(ctx, db.TransitionLiquidacionParams{
		ConsorcioID:     cid,
		ID:              pgtype.UUID{},
		EstadoActual:    liq.Estado,
		NuevoEstado:     "calculada",
		ExpectedVersion: int32(expectedVersion),
	}); err != nil {
		return LiquidacionDTO{}, err
	}

	if err := tx.Commit(ctx); err != nil {
		return LiquidacionDTO{}, err
	}

	// Obtener resultado actualizado
	dto, err := GetLiquidacion(ctx, q, consorcioID, liquidacionID)
	if err != nil {
		return LiquidacionDTO{}, err
	}

	// Persistir resultado idempotente
	if idemKey != "" {
		respJSON, _ := json.Marshal(dto)
		_ = q.UpdateIdempotencyKey(ctx, db.UpdateIdempotencyKeyParams{
			Scope:          idempScopeCalcular,
			IdempotencyKey: idemKey,
			ResponseJson:   respJSON,
		})
	}

	return dto, nil
}

func ConfirmarLiquidacion(ctx context.Context, q *db.Queries, consorcioID, liquidacionID string, expectedVersion int, idemKey string) (LiquidacionDTO, error) {
	liq, err := GetLiquidacion(ctx, q, consorcioID, liquidacionID)
	if err != nil {
		return LiquidacionDTO{}, err
	}
	if liq.Estado != "calculada" {
		return LiquidacionDTO{}, ErrLiquidacionTransicionInvalida
	}

	// Idempotencia
	if idemKey != "" {
		hash := fmt.Sprintf("%x", sha256Sum(liquidacionID+"|"+idemKey))
		inserted, err := q.InsertIdempotencyKey(ctx, db.InsertIdempotencyKeyParams{
			IdempotencyKey: idemKey,
			Scope:          idempScopeConfirmar,
			RequestHash:    hash,
			ResponseJson:   []byte("{}"),
		})
		if err != nil {
			return LiquidacionDTO{}, err
		}
		if inserted == 0 {
			existing, err := q.GetIdempotencyKey(ctx, db.GetIdempotencyKeyParams{
				Scope:          idempScopeConfirmar,
				IdempotencyKey: idemKey,
			})
			if err != nil {
				return LiquidacionDTO{}, err
			}
			if existing.RequestHash != hash {
				return LiquidacionDTO{}, ErrIdempotencyConflict
			}
			var cached LiquidacionDTO
			if err := json.Unmarshal(existing.ResponseJson, &cached); err != nil {
				return LiquidacionDTO{}, err
			}
			return cached, nil
		}
	}

	// Obtener unidades con totales del preview
	var cid pgtype.UUID
	_ = cid.Scan(consorcioID)
	unidades, err := q.GetLiquidacionUnidades(ctx, db.GetLiquidacionUnidadesParams{
		LiquidacionID: pgtype.UUID{},
	})
	if err != nil {
		return LiquidacionDTO{}, err
	}

	// Transacción: charges + outbox + estado
	tx, err := q.DB().Begin(ctx)
	if err != nil {
		return LiquidacionDTO{}, err
	}
	defer tx.Rollback(ctx)
	tq := q.WithTx(tx)

	// Generar charges y account_entries por UF
	_, _, err = cuenta_corriente.CreateCargos(ctx, tq, toUnidadCobro(unidades, liq.Periodo), "ARS")

	corrID := pgtype.UUID{}
	_ = corrID.Scan(liquidacionID)
	payloadBytes, _ := json.Marshal(LiquidacionPayload{
		LiquidacionID: liquidacionID,
		Periodo:       liq.Periodo,
		ConsorcioID:   consorcioID,
	})
	if _, err := tq.InsertOutboxEvent(ctx, db.InsertOutboxEventParams{
		CorrelationID: corrID,
		EventType:     "liquidacion.confirmada",
		Payload:       payloadBytes,
	}); err != nil {
		return LiquidacionDTO{}, err

	// Transicionar
	n, err := tq.TransitionLiquidacion(ctx, db.TransitionLiquidacionParams{
		ConsorcioID:     cid,
		ID:              pgtype.UUID{},
		EstadoActual:    "calculada",
		NuevoEstado:     "confirmada",
		ExpectedVersion: int32(expectedVersion),
	})
	if err != nil {
		return LiquidacionDTO{}, err
	}
	if n == 0 {
		return LiquidacionDTO{}, ErrLiquidacionVersionMismatch
	}

	if err := tx.Commit(ctx); err != nil {
		return LiquidacionDTO{}, err
	}

	dto, err := GetLiquidacion(ctx, q, consorcioID, liquidacionID)
	if err != nil {
		return LiquidacionDTO{}, err
	}

	if idemKey != "" {
		respJSON, _ := json.Marshal(dto)
		_ = q.UpdateIdempotencyKey(ctx, db.UpdateIdempotencyKeyParams{
			Scope:          idempScopeConfirmar,
			IdempotencyKey: idemKey,
			ResponseJson:   respJSON,
		})
	}

	return dto, nil
}

// AnularLiquidacion revierte cargos y transiciona a anulada.
func AnularLiquidacion(ctx context.Context, q *db.Queries, consorcioID, liquidacionID string, expectedVersion int) (LiquidacionDTO, error) {
	liq, err := GetLiquidacion(ctx, q, consorcioID, liquidacionID)
	if err != nil {
		return LiquidacionDTO{}, err
	}
	if liq.Estado == "anulada" || liq.Estado == "cerrada" {
		return LiquidacionDTO{}, ErrLiquidacionTransicionInvalida
	}

	var cid, lid pgtype.UUID
	_ = cid.Scan(consorcioID)
	_ = lid.Scan(liquidacionID)

	tx, err := q.DB().Begin(ctx)
	if err != nil {
		return LiquidacionDTO{}, err
	}
	defer tx.Rollback(ctx)
	tq := q.WithTx(tx)

	// Si estaba confirmada, reversar charges
	if liq.Estado == "confirmada" || liq.Estado == "publicada" {
		charges, err := tq.GetChargesByLiquidacion(ctx, db.GetChargesByLiquidacionParams{LiquidacionID: lid})
		if err == nil {
			for _, ch := range charges {
				if ch.SaldoCents > 0 {
					_, _ = cuenta_corriente.RevertirCargo(ctx, tq, ch.ID.String(), ch.UnidadID.String(), "ARS")
				}
			}
		}
	}

	n, err := tq.AnularLiquidacion(ctx, db.AnularLiquidacionParams{
		ID:              lid,
		EstadoActual:    liq.Estado,
		ExpectedVersion: int32(expectedVersion),
	})
	if err != nil {
		return LiquidacionDTO{}, err
	}
	if n == 0 {
		return LiquidacionDTO{}, ErrLiquidacionVersionMismatch
	}

	if err := tx.Commit(ctx); err != nil {
		return LiquidacionDTO{}, err
	}

	return GetLiquidacion(ctx, q, consorcioID, liquidacionID)
}

func toUnidadCobro(unidades []db.LiquidacionUnidad, periodo string) []cuenta_corriente.UnidadCobro {
	out := make([]cuenta_corriente.UnidadCobro, 0, len(unidades))
	vencimiento := fmt.Sprintf("%s-01", periodo[:4]+"-"+periodo[4:])
	for _, u := range unidades {
		out = append(out, cuenta_corriente.UnidadCobro{
			UnidadID:   u.UnidadID.String(),
			Concepto:   "expensa " + periodo,
			DueDate:    vencimiento,
			TotalCents: u.TotalCents,
		})
	}
	return out
}

func sha256Sum(s string) [32]byte {
	// Simple hash for idempotency (no crypto needed for request hash)
	var h [32]byte
	copy(h[:], s)
	return h
}

func sha256SumBytes(s string) string {
	return fmt.Sprintf("%x", sha256Sum(s))
}

	q, commit, rollback, err := h.txQueries(r)
	if err != nil {
		httpapi.WriteProblem(w, r, http.StatusUnauthorized, "unauthorized", "Token inválido", err.Error(), nil)
		return
	}
	defer func() { _ = rollback() }()

	item, err := expensas.CreateLiquidacion(r.Context(), q, chi.URLParam(r, "id"), in.Periodo, in.Vencimiento1, in.Vencimiento2)
	if err != nil {
		h.writeLiquidacionError(w, r, err)
		return
	}
	if err := commit(); err != nil {
		httpapi.WriteProblem(w, r, http.StatusInternalServerError, "internal_error", "Error interno", err.Error(), nil)
		return
	}

	h.recordAudit(r, h.consorcioAuditEvent(r, audit.Event{
		Accion:      "liquidaciones.create",
		RecursoType: "liquidacion",
		RecursoID:   item.ID,
		Diff:        map[string]any{"periodo": item.Periodo, "consorcio_id": item.ConsorcioID},
	}))

}

func (h *AuthHandlers) PatchLiquidacion(w http.ResponseWriter, r *http.Request) {
	var in struct {
		Vencimiento1 *string `json:"vencimiento_1"`
		Vencimiento2 *string `json:"vencimiento_2"`
	}
	if err := json.NewDecoder(r.Body).Decode(&in); err != nil {
		httpapi.WriteProblem(w, r, http.StatusBadRequest, "invalid_request", "Solicitud inválida", err.Error(), nil)
		return
	}
	version := parseIfMatch(r)
	if version == 0 {
		httpapi.WriteProblem(w, r, http.StatusPreconditionFailed, "if_match_required", "If-Match requerido", "Se requiere header If-Match con la versión actual", nil)
		return
	}

	q, commit, rollback, err := h.txQueries(r)
	if err != nil {
		httpapi.WriteProblem(w, r, http.StatusUnauthorized, "unauthorized", "Token inválido", err.Error(), nil)
		return
	}
	defer func() { _ = rollback() }()

	v1 := ""
	if in.Vencimiento1 != nil {
		v1 = *in.Vencimiento1
	}
	err = expensas.UpdateVencimientos(r.Context(), q, chi.URLParam(r, "consorcioId"), chi.URLParam(r, "liquidacionId"), v1, in.Vencimiento2, version)

	}

}

func parseIfMatch(r *http.Request) int {
	v := r.Header.Get("If-Match")
	if v == "" {
		return 0
	}
	n, _ := strconv.Atoi(strings.TrimSpace(v))
	return n
}

func (h *AuthHandlers) writeLiquidacionError(w http.ResponseWriter, r *http.Request, err error) {
	switch {
	case errors.Is(err, expensas.ErrLiquidacionInvalid), errors.Is(err, expensas.ErrLiquidacionTransicionInvalida):
		httpapi.WriteProblem(w, r, http.StatusBadRequest, "invalid_request", "Solicitud inválida", err.Error(), nil)
	case errors.Is(err, expensas.ErrLiquidacionNotFound):
		httpapi.WriteProblem(w, r, http.StatusNotFound, "not_found", "Recurso no encontrado", err.Error(), nil)
	case errors.Is(err, expensas.ErrLiquidacionConflict):
		httpapi.WriteProblem(w, r, http.StatusConflict, "conflict", "Conflicto", err.Error(), nil)
	case errors.Is(err, expensas.ErrLiquidacionVersionMismatch):
		httpapi.WriteProblem(w, r, http.StatusPreconditionFailed, "version_mismatch", "Versión no coincide", err.Error(), nil)
	case errors.Is(err, expensas.ErrIdempotencyConflict):
		httpapi.WriteProblem(w, r, http.StatusConflict, "idempotency_conflict", "Conflicto de idempotencia", err.Error(), nil)
	case errors.Is(err, expensas.ErrLiquidacionSinGastos), errors.Is(err, expensas.ErrLiquidacionSinUFs):
		httpapi.WriteProblem(w, r, http.StatusUnprocessableEntity, "unprocessable", "No procesable", err.Error(), nil)
	default:
		slog.Error("liquidaciones", "error", err)
		httpapi.WriteProblem(w, r, http.StatusInternalServerError, "internal_error", "Error interno", err.Error(), nil)
	}
}
